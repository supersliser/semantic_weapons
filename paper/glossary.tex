\newglossaryentry{llm}{
    name={LLM},,
    first={Large Language Models (LLMs)},
    description={A form of \gls{ai} in which a system is designed to parse massive amounts of \gls{semantic} language, simulate understanding of said language and formulate output in a semantically compatible medium}
}
\newglossaryentry{ai}{
    name={AI},
    first={Artificial Intelligence (AI)},
    description={A system which is able to simulate logical reasoning by comparing input parameters against a series of rules and logical concepts to derive a result}
}
\newglossaryentry{semantic}{
    name={Semantic},
    description={The concept of meaning and logic, usually used to describe human communication in comparision to signals sent by machines or noises created by animals. An analogue system of deriving meaning and intention from a digital format (such that individual words are digital with static definition, but sentences and paragraphs are analogue and semantic due to the possibility of interpretation)}
}
\newglossaryentry{neuralnetwork}{
    name={Neural Network},
    description={A method of building an \gls{ai} system in which data is parsed through every possible series of conections with a calculated probability to determine which pathway is most "correct"}
}
