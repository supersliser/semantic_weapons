\newglossaryentry{llm}{
    name={LLM},
    first={Large Language Model},
    description={A form of AI in which a system is designed to parse massive amounts of \gls{semantic} language, simulate understanding of said language and formulate output in a semantically compatible medium}
}
\newglossaryentry{semantic}{
    name={Semantic},
    description={The concept of meaning and logic, usually used to describe human communication in comparision to signals sent by machines or noises created by animals. An analogue system of deriving meaning and intention from a digital format (such that individual words are digital with static definition, but sentences and paragraphs are analogue and semantic due to the possibility of interpretation)}
}
\newglossaryentry{neuralnetwork}{
    name={Neural Network},
    description={A method of building an AI system in which data is parsed through every possible series of conections with a calculated probability to determine which pathway is most "correct"}
}
\newglossaryentry{dcc}{
    name={DCC},
    first={Digital Content Creation tool},
    description={A category of tools used for creating pieces of media such as video, audio, photography, etc\dots}
}
\newglossaryentry{tpu}{
    name={TPU chipset},
    description={Stands for tensor processing unit, it is a processer optimised for the parallel processing required by Artificial Intelligence}
}
\newglossaryentry{ordinal}{
    name={Ordinal},
    description={A type of data which cannot be expressed in a linear fashion, usually a set of categories or discrete tags}
}
\newglossaryentry{nominal}{
    name={Nominal},
    description={A type of data which can be expressed in a linear, contineuous fashion, usually expressed as a number with pseudo-infinite precision}
}
